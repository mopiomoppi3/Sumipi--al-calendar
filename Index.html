function renderCalGrid(){
  var person=data.persons.find(function(p){return p.id===calPersonId;});
  document.getElementById('month-label').textContent=calYear+'年 '+(calMonth+1)+'月';
  var wrap=document.getElementById('cal-grid-wrap');
  var today_s=today();
  var firstDay=new Date(calYear,calMonth,1).getDay();
  var daysInMonth=new Date(calYear,calMonth+1,0).getDate();

  var html='<div class="cal-grid">';
  var dows=['日','月','火','水','木','金','土'];
  dows.forEach(function(d){html+='<div class="cal-dow">'+d+'</div>';});

  for(var i=0;i<firstDay;i++) html+='<div class="cal-cell empty"></div>';

  for(var d=1;d<=daysInMonth;d++){
    var dateStr=calYear+'-'+pad(calMonth+1)+'-'+pad(d);
    var dow=(firstDay+d-1)%7;
    var isToday=dateStr===today_s;
    var entries=data.entries.filter(function(e){return e.personId===calPersonId&&e.date===dateStr;});
    var hasEntry=entries.length>0;
    var lastEntry=hasEntry?entries[entries.length-1]:null;
    var isMet=person&&person.met===dateStr;
    var isSince=person&&person.since===dateStr;

    var cls='cal-cell'+(dow===0?' sun':dow===6?' sat':'')+(isToday?' today':'')+(hasEntry?' has-entry':'');
    html+='<div class="'+cls+'" onclick="openDay(\''+dateStr+'\')">';
    html+='<div class="cal-date">'+d+(isMet?'<span style="color:var(--rose);margin-left:2px">💬</span>':isSince?'<span style="margin-left:2px">💗</span>':'')+'</div>';
    if(lastEntry){
      if(lastEntry.photo){
        html+='<img class="cal-thumb" src="'+lastEntry.photo+'" alt="">';
      } else {
        if(lastEntry.mood) html+='<div class="cal-mood">'+lastEntry.mood+'</div>';
        if(lastEntry.like) html+='<div class="cal-like-dot">'+heartStr(lastEntry.like)+'</div>';
      }
    }
    html+='</div>';
  }
  html+='</div>';

  // legend
  if(person){
    html+='<div style="display:flex;gap:12px;font-size:.72rem;color:var(--muted);margin-top:4px;flex-wrap:wrap">';
    if(person.since) html+='<span>💗 好きになった日</span>';
    if(person.met) html+='<span>💬 出会った日</span>';
    html+='<span>● 記録あり</span></div>';
  }
  wrap.innerHTML=html;
}

function heartStr(like){
  var n=Math.round(like/25);
  var s='';
  for(var i=0;i<n;i++) s+='♥';
  return s;
}

// ── HELPERS ────────────────────────────────────────────────────
function today(){return new Date().toISOString().split(‘T’)[0];}
function pad(n){return n<10?‘0’+n:n;}
function getDays(since){return Math.max(0,Math.floor((new Date()-new Date(since))/86400000));}
function fmtDate(d){
if(!d)return’’;
var p=d.split(’-’);
return p[0]+‘年’+parseInt(p[1])+‘月’+parseInt(p[2])+‘日’;
}
